
===== art\build-assets.mjs =====

// Builds every theme from text sprites.
//
//   art/themes/<id>/theme.json         palette (sprite letters) + CSS tokens of the theme
//   art/sprites/*.txt                  sprites shared by ALL themes (recoloured per theme)
//   art/themes/<id>/sprites/*.txt      sprites of ONE theme (replace a shared sprite of the same name)
//
// Output:
//   public/assets/sprites/<id>/<name>.png   one PNG per sprite and theme
//   src/styles/themes.css                   colour tokens + --spr-<name> url() variables per theme
//   src/lib/themes.generated.ts             theme list for the picker
//
// Sprite format: one char = one palette letter, "." = transparent.
// Animated sprites: several frames in ONE file separated by a line with ---, plus an
// optional "# order: 0 1 2 1" line (playback order, frames may repeat). The PNG is a
// horizontal strip with one cell per entry of the order.
import { existsSync, mkdirSync, readdirSync, readFileSync, writeFileSync } from "node:fs";
import { basename, dirname, join } from "node:path";
import { fileURLToPath } from "node:url";
import pngjs from "pngjs";

const { PNG } = pngjs;
const root = join(dirname(fileURLToPath(import.meta.url)), "..");
const sharedDir = join(root, "art", "sprites");
const themesDir = join(root, "art", "themes");
const outRoot = join(root, "public", "assets", "sprites");

const DEFAULT_THEME = "pink";
const LETTERS = ["a", "h", "d", "s", "l", "p", "q", "b", "y", "w", "t", "i"];

// Strip BOM (Windows PowerShell 5.1 adds one with -Encoding utf8)
const readText = (p) => readFileSync(p, "utf8").replace(/^\uFEFF/, "");
const errors = [];

function parseHex(where, key, hex) {
  const m = /^#([0-9a-f]{6})$/i.exec(String(hex));
  if (!m) {
    errors.push(`${where}: invalid color for "${key}": ${hex}`);
    return [0, 0, 0, 255];
  }
  const n = parseInt(m[1], 16);
  return [(n >> 16) & 255, (n >> 8) & 255, n & 255, 255];
}

function loadThemes() {
  const list = [];
  for (const entry of readdirSync(themesDir, { withFileTypes: true })) {
    if (!entry.isDirectory()) continue;
    const file = join(themesDir, entry.name, "theme.json");
    if (!existsSync(file)) {
      errors.push(`art/themes/${entry.name}: theme.json is missing`);
      continue;
    }
    const cfg = JSON.parse(readText(file));
    const where = `art/themes/${entry.name}/theme.json`;
    const palette = {};
    for (const letter of LETTERS) {
      if (!(letter in (cfg.sprites ?? {}))) errors.push(`${where}: sprites."${letter}" is missing`);
      else palette[letter] = parseHex(where, letter, cfg.sprites[letter]);
    }
    list.push({
      id: entry.name,
      name: cfg.name ?? entry.name,
      order: cfg.order ?? 99,
      tokens: cfg.tokens ?? {},
      palette,
    });
  }
  return list.sort((a, b) => a.order - b.order || a.id.localeCompare(b.id));
}

/** Splits a sprite file into frames and an optional playback order. */
function parseSprite(text) {
  const frames = [[]];
  let order = null;
  for (const raw of text.split(/\r?\n/)) {
    const line = raw.trimEnd();
    const m = /^#\s*order:\s*([\d\s]+)$/i.exec(line);
    if (m) {
      order = m[1].trim().split(/\s+/).map(Number);
      continue;
    }
    if (line === "---") {
      frames.push([]);
      continue;
    }
    if (line === "" || line.startsWith("#")) continue;
    frames[frames.length - 1].push(line);
  }
  return { frames, order };
}

/** name -> file path. Shared sprites first, then the theme's own (they win). */
function spriteFiles(themeId) {
  const map = new Map();
  const add = (dir) => {
    if (!existsSync(dir)) return;
    for (const f of readdirSync(dir)) if (f.endsWith(".txt")) map.set(basename(f, ".txt"), join(dir, f));
  };
  add(sharedDir);
  add(join(themesDir, themeId, "sprites"));
  return map;
}

function renderSprite(theme, name, path) {
  const where = path.replace(root, "").replace(/^[\\/]/, "");
  const { frames, order: rawOrder } = parseSprite(readText(path));
  const before = errors.length;

  if (frames.some((f) => f.length === 0)) {
    errors.push(`${where}: empty sprite or empty frame`);
    return;
  }
  const width = frames[0][0].length;
  const height = frames[0].length;
  frames.forEach((rows, i) => {
    if (rows.length !== height) {
      errors.push(`${where}: frame ${i} has ${rows.length} rows, expected ${height}`);
      return;
    }
    rows.forEach((row, y) => {
      if (row.length !== width) {
        errors.push(`${where}: frame ${i} row ${y + 1} has ${row.length} chars, expected ${width}`);
        return;
      }
      for (const ch of row) {
        if (ch !== "." && !theme.palette[ch]) {
          errors.push(`${where}: frame ${i} row ${y + 1} uses unknown letter "${ch}"`);
          break;
        }
      }
    });
  });
  const order = rawOrder ?? frames.map((_, i) => i);
  if (order.some((i) => !Number.isInteger(i) || i < 0 || i >= frames.length)) {
    errors.push(`${where}: "# order:" refers to a frame that does not exist`);
  }
  if (errors.length > before) return;

  const png = new PNG({ width: width * order.length, height });
  order.forEach((frameIndex, k) => {
    frames[frameIndex].forEach((row, y) => {
      [...row].forEach((ch, x) => {
        const i = (y * png.width + k * width + x) * 4;
        const [r, g, b, a] = ch === "." ? [0, 0, 0, 0] : theme.palette[ch];
        png.data[i] = r;
        png.data[i + 1] = g;
        png.data[i + 2] = b;
        png.data[i + 3] = a;
      });
    });
  });
  writeFileSync(join(outRoot, theme.id, `${name}.png`), PNG.sync.write(png));
}

const themes = loadThemes();
const generated = []; // { theme, names }

for (const theme of themes) {
  mkdirSync(join(outRoot, theme.id), { recursive: true });
  const files = spriteFiles(theme.id);
  for (const [name, path] of [...files].sort(([a], [b]) => a.localeCompare(b))) renderSprite(theme, name, path);
  generated.push({ theme, names: [...files.keys()].sort() });
  console.log(`ok  ${theme.id}  ${files.size} sprites`);
}

if (errors.length > 0) {
  console.error("\nErrors:\n" + errors.map((e) => "  - " + e).join("\n"));
  process.exitCode = 1;
} else {
  // ---- themes.css ----
  let css = "/* GENERATED by art/build-assets.mjs - do not edit. */\n\n";
  for (const { theme, names } of generated) {
    css += theme.id === DEFAULT_THEME ? `:root,\n:root[data-theme="${theme.id}"] {\n` : `:root[data-theme="${theme.id}"] {\n`;
    for (const [k, v] of Object.entries(theme.tokens)) css += `  --${k}: ${v};\n`;
    for (const name of names) css += `  --spr-${name}: url("/assets/sprites/${theme.id}/${name}.png");\n`;
    css += "}\n\n";
  }
  writeFileSync(join(root, "src", "styles", "themes.css"), css);

  // ---- themes.generated.ts ----
  const list = themes.map((t) => ({
    id: t.id,
    name: t.name,
    swatches: [t.tokens.bg, t.tokens.accent, t.tokens.ink],
  }));
  const ts =
    "// GENERATED by art/build-assets.mjs - do not edit.\n" +
    `export const THEMES = ${JSON.stringify(list, null, 2)} as const;\n\n` +
    'export type ThemeId = (typeof THEMES)[number]["id"];\n';
  mkdirSync(join(root, "src", "lib"), { recursive: true });
  writeFileSync(join(root, "src", "lib", "themes.generated.ts"), ts);
}

===== src\lib\theme.ts =====

import { THEMES, type ThemeId } from "./themes.generated";

export { THEMES };
export type { ThemeId };

export const DEFAULT_THEME: ThemeId = "pink";
const STORAGE_KEY = "cupid-theme";

export function isTheme(value: unknown): value is ThemeId {
  return THEMES.some((t) => t.id === value);
}

/** Sets the theme on <html> (CSS tokens and sprite variables switch instantly). */
export function applyTheme(id: ThemeId): void {
  document.documentElement.dataset.theme = id;
  try {
    window.localStorage.setItem(STORAGE_KEY, id);
  } catch {
    /* storage unavailable: the theme is still applied for this session */
  }
}

/** Last used theme, read synchronously so the first paint already has it. */
export function initialTheme(): ThemeId {
  try {
    const stored = window.localStorage.getItem(STORAGE_KEY);
    return isTheme(stored) ? stored : DEFAULT_THEME;
  } catch {
    return DEFAULT_THEME;
  }
}

import type { CSSProperties } from "react";
import { useSettingsStore } from "../state/settingsStore";

type AnimSpriteProps = {
  /** Strip name in /assets/sprites/<theme> (without .png). */
  name: string;
  /** Size of ONE frame, in logical px. */
  w: number;
  h: number;
  /** Number of cells in the strip (the length of its "# order:"). */
  frames: number;
  fps?: number;
  /** Seconds before the loop starts: use it to desynchronise copies. */
  delay?: number;
  mirror?: boolean;
  className?: string;
};

export function AnimSprite({
  name,
  w,
  h,
  frames,
  fps = 8,
  delay = 0,
  mirror = false,
  className = "",
}: AnimSpriteProps) {
  const theme = useSettingsStore((s) => s.theme);
  const style = {
    width: `calc(var(--s) * ${w}px)`,
    height: `calc(var(--s) * ${h}px)`,
    backgroundImage: `url(/assets/sprites/${theme}/${name}.png)`,
    transform: mirror ? "scaleX(-1)" : undefined,
    animationDuration: `${frames / fps}s`,
    animationDelay: delay ? `${delay}s` : undefined,
    "--sheet-w": w * frames,
    "--frames": frames,
  } as CSSProperties;

  return <span className={`px-anim ${className}`.trim()} style={style} aria-hidden="true" />;
}

import { useEffect, useState, type CSSProperties } from "react";
import { useTranslation } from "react-i18next";
import { PixelButton } from "./PixelButton";
import { PixelInput } from "./PixelInput";
import { Sprite } from "./Sprite";

type Phase = "idle" | "running" | "done";

const TOTAL = 8; // mock: replaced by real progress in H7

export function ImportBox() {
  const { t } = useTranslation();
  const [link, setLink] = useState("");
  const [phase, setPhase] = useState<Phase>("idle");
  const [done, setDone] = useState(0);

  useEffect(() => {
    if (phase !== "running") return;
    const id = window.setTimeout(() => {
      if (done + 1 >= TOTAL) {
        setDone(TOTAL);
        setPhase("done");
      } else {
        setDone(done + 1);
      }
    }, 350);
    return () => window.clearTimeout(id);
  }, [phase, done]);

  const start = () => {
    setDone(0);
    setPhase("running");
  };

  const status =
    phase === "running"
      ? t("import.progress", { done, total: TOTAL })
      : phase === "done"
        ? t("import.done")
        : t("import.idle");

  return (
    <div className="import">
      <PixelInput
        placeholder={t("import.placeholder")}
        aria-label={t("import.placeholder")}
        value={link}
        onChange={(e) => setLink(e.target.value)}
        autoComplete="off"
        spellCheck={false}
      />
      <PixelButton onClick={start} disabled={phase === "running" || link.trim() === ""}>
        {t("import.button")}
      </PixelButton>
      <div
        className="px-progress import__bar"
        style={{ "--p": done / TOTAL } as CSSProperties}
        role="progressbar"
        aria-label={t("import.button")}
        aria-valuemin={0}
        aria-valuemax={100}
        aria-valuenow={Math.round((done / TOTAL) * 100)}
      >
        <div className="px-progress__track" />
        {done > 0 && <div className="px-progress__fill" />}
        {phase !== "idle" && (
          <div className="px-progress__star">
            <Sprite name="knob-download" w={9} h={8} />
          </div>
        )}
      </div>
      <div className="import__status ellipsis" aria-live="polite">
        {status}
      </div>
    </div>
  );
}

===== src\components\ProgressStar.tsx =====

import { useRef, useState, type CSSProperties, type KeyboardEvent, type PointerEvent } from "react";
import { Sprite } from "./Sprite";

export type Knob = { name: string; dragName?: string; w: number; h: number };

export const KNOB_MUSIC: Knob = { name: "knob-music", dragName: "knob-music-drag", w: 11, h: 11 };
export const KNOB_VOLUME: Knob = { name: "knob-volume", w: 9, h: 8 };

type ProgressStarProps = {
  /** 0..1 */
  value: number;
  onChange?: (value: number) => void;
  /** Called when the drag ends or a key is pressed (use it to seek). */
  onCommit?: (value: number) => void;
  step?: number;
  label: string;
  disabled?: boolean;
  knob?: Knob;
};

/** Height of the bar container in logical px (see .px-progress). */
const BAR_H = 11;

const clamp = (v: number) => Math.min(1, Math.max(0, v));

export function ProgressStar({
  value,
  onChange,
  onCommit,
  step = 0.02,
  label,
  disabled = false,
  knob = KNOB_MUSIC,
}: ProgressStarProps) {
  const ref = useRef<HTMLDivElement>(null);
  const [dragging, setDragging] = useState(false);
  const p = clamp(value);

  // Free space at each end = half the knob, so the knob stops before the edge.
  const pad = Math.ceil(knob.w / 2);
  const knobTop = Math.floor((BAR_H - knob.h) / 2);
  const knobHalf = Math.floor(knob.w / 2);

  const ratioFrom = (e: PointerEvent<HTMLDivElement>) => {
    const el = ref.current;
    if (!el) return value;
    const r = el.getBoundingClientRect();
    const padPx = (pad * r.height) / BAR_H; // = pad * --s
    return clamp((e.clientX - r.left - padPx) / (r.width - 2 * padPx));
  };

  const onPointerDown = (e: PointerEvent<HTMLDivElement>) => {
    if (disabled) return;
    e.currentTarget.setPointerCapture(e.pointerId);
    setDragging(true);
    onChange?.(ratioFrom(e));
  };
  const onPointerMove = (e: PointerEvent<HTMLDivElement>) => {
    if (dragging) onChange?.(ratioFrom(e));
  };
  const onPointerUp = (e: PointerEvent<HTMLDivElement>) => {
    if (!dragging) return;
    setDragging(false);
    onCommit?.(ratioFrom(e));
  };

  const onKeyDown = (e: KeyboardEvent<HTMLDivElement>) => {
    if (disabled) return;
    let next: number | null = null;
    if (e.key === "ArrowRight" || e.key === "ArrowUp") next = clamp(value + step);
    else if (e.key === "ArrowLeft" || e.key === "ArrowDown") next = clamp(value - step);
    else if (e.key === "Home") next = 0;
    else if (e.key === "End") next = 1;
    if (next === null) return;
    e.preventDefault();
    onChange?.(next);
    onCommit?.(next);
  };

  return (
    <div
      ref={ref}
      className="px-progress"
      style={{ "--p": p, "--pad": pad } as CSSProperties}

    >

import type { HTMLAttributes } from "react";

type PixelPanelProps = HTMLAttributes<HTMLDivElement> & {
  /** Adds the four corner ornaments. */
  deco?: boolean;
};

export function PixelPanel({ className = "", deco = false, children, ...rest }: PixelPanelProps) {
  const cls = ["px-panel", deco ? "px-panel--deco" : "", className].filter(Boolean).join(" ");
  return (
    <div className={cls} {...rest}>
      {deco && (
        <>
          <i className="px-corner px-corner--tl" aria-hidden="true" />
          <i className="px-corner px-corner--tr" aria-hidden="true" />
          <i className="px-corner px-corner--bl" aria-hidden="true" />
          <i className="px-corner px-corner--br" aria-hidden="true" />
        </>
      )}
      {children}
    </div>
  );
}
